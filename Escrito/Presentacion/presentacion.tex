\section{Motivación y objetivos}

\begin{frame}{El SEN en transición}
    \begin{itemize}
        \item El Sistema Eléctrico Nacional (SEN) chileno ha experimentado una transición acelerada hacia generación solar: la participación ERNC pasó de $\sim$20\% a $\sim$40\% de la generación total entre 2019 y 2024.
        \item Esto introduce \textbf{volatilidad estructural creciente} en el costo marginal horario: episodios de sobreoferta solar (\textit{curtailment}) y transiciones de precio abruptas al entrar y salir el bloque fotovoltaico.
        \item En 2024 se perdieron \textbf{5\,908.71\,GWh} de energía eólica y solar por \textit{curtailment}, un aumento del \textbf{149\%} respecto al año anterior.
        \item Los métodos estadísticos clásicos (ARIMA, modelos de descomposición) capturan mal esta dinámica no lineal y cambiante.
        \item \textbf{Pregunta}: ¿puede un modelo híbrido de \textit{deep learning}, con una señal exógena simple y determinista, mejorar el pronóstico horario y cuantificar honestamente su incertidumbre?
    \end{itemize}
\end{frame}

\begin{frame}{Objetivos}
    \textbf{Objetivo general}
    \begin{itemize}
        \item Desarrollar y comparar dos estrategias híbridas (P+C y S+C) para predecir el costo marginal del SEN con el Temporal Fusion Transformer (TFT), con cuantificación de incertidumbre mediante Conformal Prediction.
    \end{itemize}
    \vspace{0.5em}
    \textbf{Objetivos específicos}
    \begin{enumerate}
        \small
        \item Caracterizar el costo marginal del SEN (estacionalidad, tendencia, volatilidad).
        \item Diseñar las estrategias de \textit{features} P+C (Prophet) y S+C (posición solar).
        \item Comparar LayerNorm y Dynamic Tanh con pruebas estadísticas pareadas.
        \item Combinar Prophet y el TFT mediante Stacking Optimization dentro de P+C.
        \item Comparar P+C y S+C con Diebold-Mariano (varianza Newey-West).
        \item Implementar Conformal Prediction (variantes clásicas y LW-ACP) y validar cobertura y eficiencia.
        \item Evaluar variables hídricas propias de Puerto Montt.
    \end{enumerate}
\end{frame}

\section{Datos}

\begin{frame}{Datos}
    \begin{itemize}
        \item Fuente: Coordinador Eléctrico Nacional (CEN) de Chile.
        \item Variable objetivo: costo marginal horario, enero 2020 -- abril 2026 ($\sim$55\,000 obs.\ por barra).
        \item Ocho barras de 220\,kV, seleccionadas por representatividad geográfica:

\begin{frame}{Dos estrategias de \textit{features} exógenas}
    \begin{columns}
        \column{0.48\textwidth}
        \textbf{P+C: Prophet + Clima}
        \begin{itemize}
            \item Prophet descompone la serie en tendencia + estacionalidad.
            \item El TFT modela tanto el precio como el residuo de Prophet.
            \item Stacking Optimization combina ambas predicciones.
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{S+C: Sol + Clima}
        \begin{itemize}
            \item Reemplaza la descomposición de Prophet por variables de posición solar (elevación, coseno de elevación) calculadas de forma \textbf{determinista} vía \texttt{pvlib}.
            \item Una única instancia del TFT, sin Stacking.
            \item Más simple: no requiere ajustar Prophet ni un segundo modelo de residuos.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Temporal Fusion Transformer}
    \begin{itemize}
        \item Arquitectura de \textit{deep learning} para forecasting de series de tiempo con atención interpretable (Lim et al., 2021).
        \item Ventana de contexto: 168 horas (1 semana). Horizonte principal: $h=1$ hora.
        \item Se entrena minimizando MAE directamente (no la pérdida cuantílica original del TFT), dejando la cuantificación de incertidumbre a Conformal Prediction.
        \item Se comparan dos mecanismos de normalización interna:
        \begin{itemize}
            \item \textbf{LayerNorm (LN)}: normalización local, por paso temporal.
            \item \textbf{Dynamic Tanh (DyT)}: un único escalar $\alpha$ global aprendido, sin normalización explícita.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Conformal Prediction: de Split CP a LW-ACP}
    \centering
    \textbf{Split CP} $\;\longrightarrow\;$ \textbf{ACI} $\;\longrightarrow\;$ \textbf{LW-ACP}
    \vspace{1em}
    \begin{itemize}
        \item \textbf{Split CP} (clásico): un cuantil fijo $q$, calculado una vez sobre calibración.
        \item \textbf{ACI} (Gibbs \& Candès, 2021): adapta el nivel de cobertura $\alpha_t$ \textit{online} según los aciertos y fallos recientes.
        \item \textbf{LW Split CP}: normaliza el \textit{score} de no conformidad por la desviación estándar de cada hora del día $\hat\sigma_h$ (heterocedasticidad horaria).
        \item \textbf{LW-ACP}: ACI aplicado sobre el \textit{score} ponderado por hora. No es un método nuevo; permite separar el aporte de cada mecanismo (ponderación horaria y adaptación \textit{online}).
    \end{itemize}
\end{frame}

\section{Resultados}

\begin{frame}{S+C supera a P+C}
    \begin{table}
        \centering
        \small
        \begin{tabular}{lcc}
            \toprule
             & \textbf{S+C mejor} & \textbf{Significativo (DM, Holm)} \\
            \midrule
            De 16 combinaciones (8 barras $\times$ 2 arq.) & 14 & 9 \\
            \bottomrule
        \end{tabular}
    \end{table}
    \vspace{1em}
    \begin{itemize}
        \item S+C, la estrategia más simple, supera a P+C en 14 de 16 combinaciones; 9 son significativas tras la corrección de Holm-Bonferroni (Diebold-Mariano con varianza Newey-West), todas a favor de S+C.
        \item La elección se repite en validación: S+C obtiene menor MAE en 15 de 16 combinaciones.
        \item En el encoder del TFT, la posición solar recibe más peso que el clima en 7 de las 8 barras.
        \item Conclusión: a una hora, la posición solar es al menos tan informativa como la descomposición de Prophet, con un pipeline mucho más simple.
    \end{itemize}
\end{frame}

\begin{frame}{LayerNorm vs.\ Dynamic Tanh}
    \begin{table}
        \centering
        \small
        \begin{tabular}{lccc}
            \toprule
            & \textbf{Significativas (DM, Holm)} & \textbf{LN gana} & \textbf{Dif.\ típica} \\
            \midrule
            De 16 combinaciones & 6 & 6 de 6 & $<0.5$\,USD/MWh \\
            \bottomrule
        \end{tabular}
    \end{table}
    \vspace{0.5em}
    \begin{itemize}
        \item A una hora LN gana las 6 combinaciones significativas tras Holm-Bonferroni, pero la ventaja \textbf{no supera la variabilidad entre semillas ni se mantiene a 24 horas}: en la práctica, ambas son equivalentes.
        \item \textbf{Robustez a la semilla} (5 semillas, 3 barras piloto): la diferencia entre arquitecturas (0.13--0.21\,USD/MWh) es del orden de la desviación entre semillas (0.06--0.30), y ninguna es significativa tras Holm-Bonferroni (test $t$ de Welch).
        \begin{itemize}
            \item No contradice al test DM: ese compara dos modelos entrenados concretos hora a hora, no la variabilidad de volver a entrenar cada arquitectura.
        \end{itemize}
        \item \textbf{Hipótesis de los \textit{spikes}}: DyT no es más preciso en horas de precio alto, pero sí en las de salto brusco (significativo en 3 barras).
    \end{itemize}
\end{frame}

\begin{frame}{Análisis de errores extremos: desfase de fase}
    \begin{itemize}
        \item Fuera del apagón del 25 de febrero de 2025, la mayoría de los veinte mayores errores de validación de cada barra (entre 11 y 19) corresponden a un \textbf{desfase de fase}: el modelo predice correctamente la magnitud de la transición, pero con una a dos horas de retraso.
        \item Patrón consistente en todo el SEN, no un artefacto de una barra particular.
        \item Control: el mismo criterio aplicado a la persistencia de 24 horas clasifica como desfase solo el 16\% de sus mayores errores, frente al 51\% del TFT.
        \item Errores de gran amplitud pero muy localizados en el tiempo $\Rightarrow$ motiva directamente el valor de Conformal Prediction adaptativo y ponderado por hora.
    \end{itemize}
\end{frame}

\begin{frame}{Puerto Montt: un caso no resuelto}
    \begin{itemize}
        \item Peor desempeño del sistema de forma consistente: MAE de prueba 13.68\,USD/MWh vs.\ 6.6--8.3\,USD/MWh del resto.
        \item Media y desviación estándar de precio sustancialmente mayores: dependencia hidrológica de baja frecuencia (meses), más lenta que la ventana de contexto del TFT (168\,h).
        \item Se probaron \textit{features} hídricas propias (cota del embalse, agua caída) en variantes H+C y S+H+C: \textbf{no lograron mejorar} sobre S+C.
        \item Puerto Montt también exhibe desfase de fase, pero casi nunca en las horas de transición solar (1 de 19 casos, frente a 21 de 63 en el resto del SEN).
        \item Queda documentado como un caso estructuralmente distinto, no resuelto por las extensiones exploradas.
    \end{itemize}
\end{frame}

        \item Ningún método domina las cinco métricas; LW-ACP gana en Winkler (que combina ancho y calibración) y en pinball superior.

\begin{frame}{LW-ACP robusto a colas pesadas}
    \footnotesize
    \begin{itemize}
        \item LW-ACP escala cada hora con la \textbf{desviación estándar} de su error, con un piso igual a la global. Las colas pesadas la inflan: en Atacama a las 15\,h, P90 del error 0.05\,USD/MWh, desviación 16.3.
        \item Resultado: sobrecubre al mediodía y subcubre en las transiciones solares.
        \item \textbf{Variante}: escalar cada hora por un \textbf{cuantil} de su error (P90, con piso de 15\,USD/MWh, ambos elegidos en validación). Se acerca a calibrar cada hora por separado (Mondrian CP).
    \end{itemize}
    \begin{table}
        \centering
        \begin{tabular}{lrrrr}
            \toprule
            90\% nominal & \textbf{Ancho} & \textbf{Winkler} & \textbf{SD hora} & \textbf{Transición} \\
            \midrule
            LW-ACP          & 41.98 & 71.87 & 0.071 & 0.799 \\
            LW-ACP robusto  & \textbf{40.40} & \textbf{69.15} & \textbf{0.039} & \textbf{0.856} \\
            \bottomrule
        \end{tabular}
    \end{table}
    \begin{itemize}
        \item Mejor Winkler en las 8 barras y el más angosto de todos los métodos; misma estabilidad mes a mes. Al 95\% también mejora (87.75 vs.\ 89.89).
        \item A diferencia de la versión principal, supera significativamente a ACI en las 8 barras al 90\% (Diebold-Mariano, Holm-Bonferroni).
        \item A 24 horas no agrega: la adaptación por hora ya ajusta cada hora. Calibración móvil e intervalo asimétrico: sin mejora.
    \end{itemize}
\end{frame}

\section{Horizonte de pronóstico}

\begin{frame}{Alcance: una hora como escenario controlado}
    \begin{itemize}
        \item Todos los resultados principales pronostican \textbf{la hora siguiente}, usando el precio observado hasta la hora anterior.
        \item Es el mismo horizonte del antecedente chileno más cercano (León et al., 2026), que deja el \textit{day-ahead} como trabajo futuro.
        \item Sirve como \textbf{escenario controlado} para comparar estrategias de \textit{features}, arquitecturas y métodos de Conformal Prediction.
        \item Pero el horizonte de decisión más relevante del mercado es el \textit{day-ahead}: por eso se evaluó también el modelo a \textbf{24 horas}.
        \item La comparación con el CEN cambia con el horizonte: a una hora el modelo usa información que el programado no tiene; a igual anticipación, gana en dos de seis barras.
    \end{itemize}
\end{frame}

    \centering
    \includegraphics[width=0.92\textwidth]{fig_h24_mae_ancho.pdf}

\begin{frame}{¿Por qué P+C mejora a 24 horas?}
    \footnotesize
    \begin{columns}[T]

    \vspace{0.4em}
    \begin{itemize}
        \item A 1 hora mandan el precio reciente y la hora del día, que ambas estrategias tienen: S+C gana por simplicidad.
        \item A 24 horas el precio reciente pierde peso y P+C recurre al día de la semana, que solo él conoce.
        \item Propuesta: un único TFT con posición solar + \texttt{trend}, \texttt{weekly} y feriado.
    \end{itemize}
\end{frame}

\begin{frame}{Extensiones exploradas}
    \begin{itemize}
        \item \textbf{Piloto sin recorte de outliers} ($h=1$, ocho barras): no perjudica el MAE general (mejora significativa en Pan de Azúcar y Quillota tras Holm-Bonferroni, ninguna en contra) y reduce levemente los errores más grandes (MAE en el percentil 99: $-8\%$; Winkler de LW-ACP: $-1.3\%$).
        \item \textbf{Modelo global} (un TFT para las ocho barras): no supera al modelo por barra, y el efecto del recorte no se replica en él.
        \item \textbf{Sensibilidad a pérdida asimétrica}: la ventaja de S+C-LN sobre la persistencia se sostiene en 7 de 8 barras independientemente de qué error sea más costoso; en Puerto Montt depende de la asimetría.
    \end{itemize}
\end{frame}

% ============================================================
\section{Conclusiones}

\begin{frame}{Conclusiones principales}
    \small
    \begin{enumerate}
        \item \textbf{La estrategia más simple es la más precisa a una hora}: S+C supera a P+C (9 de 16 diferencias significativas tras Holm-Bonferroni, ninguna a favor de P+C); a 24 horas P+C es algo más preciso (4.5\%), y S+C queda cerca con un solo modelo.
        \item \textbf{LayerNorm y DyT son equivalentes}: la ventaja de LN a una hora no supera la variabilidad entre semillas ni se mantiene a 24 horas.
        \item \textbf{En LW-ACP domina la adaptación \textit{online}}: explica casi toda la ganancia sobre Split CP; la ponderación horaria aporta poco con la desviación estándar, pero con una escala robusta supera a ACI en las 8 barras.
        \item Los errores más extremos siguen un patrón mecanístico común (desfase de fase de una a dos horas), no ruido aleatorio disperso.
        \item A una hora supera a la literatura reciente de ML clásico (comparación aproximada); a 24 horas la calibración de los intervalos se mantiene, pero la predicción puntual no supera a la persistencia.
        \item Puerto Montt permanece como un caso no resuelto, estructuralmente distinto al resto del SEN.
    \end{enumerate}
\end{frame}

\begin{frame}{Limitaciones}
    \begin{itemize}
        \item \textbf{Horizonte}: los resultados principales son a una hora; a 24 horas la predicción puntual no supera a la persistencia diaria.
        \item Un único split cronológico entrenamiento/validación/prueba (no \textit{walk-forward validation}) por costo computacional.
        \item El conjunto de prueba no ajustó parámetros, pero sí motivó qué extensiones explorar (LW-ACP robusto, $\alpha_t$ por hora, variables hídricas).
        \item 30 de los 48 entrenamientos a una hora alcanzaron el tope de 100 épocas, más en P+C que en S+C.
        \item Sin semilla fija para los modelos principales del Capítulo 4 (sí para el estudio de robustez).
        \item Esfuerzo de ajuste de hiperparámetros asimétrico entre Prophet (optimizado por barra) y TFT (fijo).
        \item Pipeline no directamente desplegable en operación real: ERA5 se publica con rezago.
    \end{itemize}
\end{frame}

\begin{frame}{Trabajo futuro}
    \begin{itemize}
        \item Mejorar el \textit{day-ahead} con información conocida de antemano, por ejemplo el costo marginal programado del CEN como variable de entrada.
        \item Incorporar factores técnicos, de seguridad y propios de la operación del sistema.
        \item Un TFT con posición solar más \texttt{trend}, \texttt{weekly} y feriado, sin Stacking.
        \item Comparar LW-ACP contra AgACI (agregación de múltiples $\gamma$ vía expertos \textit{online}).
        \item Adoptar la escala robusta como versión principal de LW-ACP y combinarla con niveles $\alpha_t$ por hora que compartan información entre horas.
        \item \textit{Walk-forward validation} sobre múltiples ventanas rodantes.
        \item Investigar directamente el mecanismo del desfase de fase (pérdida que penalice el error de temporización).
        \item Evaluar en Puerto Montt indicadores de congestión del corredor Charrúa--Puerto Montt.
    \end{itemize}
\end{frame}
